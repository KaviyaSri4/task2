1. find the duplicate number:

import java.util.*;

class Main {
    public static void main(String[] args)
    {
        Scanner obj = new Scanner(System.in);

        int num = obj.nextInt();
        int[] arr = new int[num];

        // Read array elements
        for (int i = 0; i < num; i++)
        {
            arr[i] = obj.nextInt();
        }

        // Find duplicate elements
        for (int i = 0; i < num; i++)
        {
            for (int j = i + 1; j < num; j++)
            {
                if (arr[i] == arr[j])   // != 
                {
                    System.out.println(arr[i]);
                }
            }
        }
    }
}

output:

5
1 2 3 2 4

2

2.Find the Unique elements


import java.util.*;

class Main {
    public static void main(String[] args)
    {
        Scanner obj = new Scanner(System.in);

        int num = obj.nextInt();
        int[] arr = new int[num];

        // Read array elements
        for (int i = 0; i < num; i++)
        {
            arr[i] = obj.nextInt();
        }

        // Find unique elements
        for (int i = 0; i < num; i++)
        {
           
             int count=0; 
           
            for (int j = 0; j < num; j++)
            {
                if (arr[i] == arr[j])   
                {
                    count++;
                }
            }
                 
            if(count == 1)
            {
              System.out.println(arr[i]);   // System.out.println(arr[i] "-->" +count)  // print count
            }         

        }
    }
}

6
10 20 30 20 40 10

30 40
